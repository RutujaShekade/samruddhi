# 8 Queens Problem using Backtracking

N = 8

# Create chessboard
board = [[0 for _ in range(N)] for _ in range(N)]


# Check whether position is safe
def is_safe(row, col):

    # Check same column
    for i in range(row):
        if board[i][col] == 1:
            return False

    # Check upper left diagonal
    i = row - 1
    j = col - 1

    while i >= 0 and j >= 0:
        if board[i][j] == 1:
            return False
        i -= 1
        j -= 1

    # Check upper right diagonal
    i = row - 1
    j = col + 1

    while i >= 0 and j < N:
        if board[i][j] == 1:
            return False
        i -= 1
        j += 1

    return True


# Backtracking function
def solve(row):

    # All 8 queens are placed
    if row == N:
        return True

    # Try every column
    for col in range(N):

        if is_safe(row, col):

            # Place queen
            board[row][col] = 1

            # Solve next row
            if solve(row + 1):
                return True

            # Backtrack
            board[row][col] = 0

    return False


# Execute program
if solve(0):

    print("Solution Found!\n")

    for row in board:
        for col in row:
            if col == 1:
                print("Q", end=" ")
            else:
                print(".", end=" ")
        print()

else:
    print("No Solution Found")
