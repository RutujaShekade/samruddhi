import random

# Distance between cities
dist = [
    [0, 10, 15, 20, 25],
    [10, 0, 35, 25, 17],
    [15, 35, 0, 30, 28],
    [20, 25, 30, 0, 16],
    [25, 17, 28, 16, 0]
]

# Calculate total distance
def cost(route):
    total = 0

    for i in range(len(route)):
        total += dist[route[i]][route[(i+1) % len(route)]]

    return total


# Generate neighbouring routes
def neighbors(route):
    result = []

    for i in range(1, len(route)):
        for j in range(i+1, len(route)):
            new_route = list(route)
            new_route[i], new_route[j] = new_route[j], new_route[i]
            result.append(tuple(new_route))

    return result


# Local Beam Search
def local_beam_search(k=3):

    random.seed(1)

    cities = [1, 2, 3, 4]

    beam = [
        (0,) + tuple(random.sample(cities, 4))
        for _ in range(k)
    ]

    best = min(beam, key=cost)

    for _ in range(20):

        candidates = set()

        for route in beam:
            candidates.update(neighbors(route))

        beam = sorted(candidates, key=cost)[:k]

        current = min(beam, key=cost)

        if cost(current) < cost(best):
            best = current

    return best, cost(best)


# Run program
route, distance = local_beam_search()

cities = ['A', 'B', 'C', 'D', 'E']

print("Best Route:")

for city in route:
    print(cities[city], end=" -> ")

print(cities[route[0]])

print("Minimum Distance:", distance)
